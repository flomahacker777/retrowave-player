// Подключение в Compose: Modifier.graphicsLayer { renderEffect = ... }

uniform shader composable;
uniform float2 iResolution;
uniform float  iTime;
uniform float2 iTouchPos;    // Нормализованная позиция пальца (-1..1)
uniform float  iTouchPress;  // Плавное нажатие (0.0 .. 1.0)
uniform float  iBevelScale;  // Толщина фаски акрила (default 1.0)

float sdRoundBox(float2 p, float2 b, float r) {
    float2 q = abs(p) - b + r;
    return length(max(q, 0.0)) + min(max(q.x, q.y), 0.0) - r;
}

half4 main(float2 fragCoord) {
    float2 uv = fragCoord / iResolution;
    float aspect = iResolution.x / iResolution.y;
    float2 p = (uv - 0.5) * float2(aspect, 1.0);

    float2 halfSize = float2(0.34 * aspect, 0.34);
    float d = sdRoundBox(p, halfSize, 0.045);

    if (d > 0.0) {
        return composable.eval(fragCoord);
    }

    // 45° фаска акрилового блока + вязкая линза под пальцем
    float bevelW = 0.042 * iBevelScale;
    float edgeMask = smoothstep(-bevelW, 0.0, d);
    float2 grad = normalize(p + 1e-5);

    float distFinger = length(p - iTouchPos);
    float lens = exp(-distFinger * distFinger * 18.0) * iTouchPress;
    float2 lensOffset = (p - iTouchPos) * lens * 0.14;

    // Хроматическая дисперсия PMMA (R / G / B)
    float2 baseRefract = grad * pow(edgeMask, 1.8) * 0.048 * iBevelScale + lensOffset;
    half r = composable.eval((uv - baseRefract * 1.12) * iResolution).r;
    half g = composable.eval((uv - baseRefract * 1.00) * iResolution).g;
    half b = composable.eval((uv - baseRefract * 0.88) * iResolution).b;

    half3 col = half3(r, g, b);
    // Молочная диффузия под чёрным текстом + торцевой циановый оттенок PMMA
    col = mix(col, half3(0.95, 0.97, 0.99), 0.34);
    col += half3(0.04, 0.14, 0.13) * pow(edgeMask, 2.4);

    // Студийный блик на фаске
    float spec = pow(max(0.0, dot(grad, normalize(float2(-0.6, -0.8)))), 6.0) * edgeMask;
    col += half3(spec * 0.65);

    return half4(col, 1.0);
}</pre>

      // 4: Swiss Graphic Poster (Reference 80636.jpg / 80634.jpg)
      vec3 paper = vec3(0.86, 0.88, 0.90);
      float circle = smoothstep(0.245, 0.24, length(p - vec2(0.0, 0.03)));
      vec3 col = mix(paper, vec3(0.14, 0.16, 0.19), circle);
      float stripe = step(0.5, fract((p.x + p.y) * 18.0)) * smoothstep(0.18, 0.175, length(p - vec2(0.08, -0.14)));
      col = mix(col, vec3(0.94, 0.96, 0.98), stripe * 0.85);
      return col;
    }
  }

  // Bayer 4x4 threshold for Mac 1-Bit dithering
  float bayer4(vec2 frag) {
    ivec2 ip = ivec2(mod(frag, 4.0));
    int idx = ip.y * 4 + ip.x;
    const float m[16] = float[16](
       0.0/16.0,  8.0/16.0,  2.0/16.0, 10.0/16.0,
      12.0/16.0,  4.0/16.0, 14.0/16.0,  6.0/16.0,
       3.0/16.0, 11.0/16.0,  1.0/16.0,  9.0/16.0,
      15.0/16.0,  7.0/16.0, 13.0/16.0,  5.0/16.0
    );
    return m[idx];
  }

  // Apply chosen retro filter inside the acrylic glass volume
  vec3 applyRetroEffect(vec3 col, vec2 uvBlock, vec2 fragCoord, int mode) {
    if (mode == 0) {
      // 0: Y2K CCD (2003 compact digital camera sensor: RGGB Bayer micro-grid, highlight vertical smear, 8x8 DCT shimmer)
      ivec2 px = ivec2(mod(floor(fragCoord * 0.5), 2.0));
      vec3 bayerTint = vec3(0.96, 1.02, 1.05);
      if (px.x == 0 && px.y == 0) bayerTint = vec3(1.06, 0.98, 0.96);
      else if (px.x == 1 && px.y == 1) bayerTint = vec3(0.95, 0.98, 1.08);
      else bayerTint = vec3(0.97, 1.04, 0.98);

      // Vertical CCD highlight smear
      float luma = dot(col, vec3(0.299, 0.587, 0.114));
      float smear = smoothstep(0.68, 0.98, luma) * 0.11;
      col += vec3(0.75, 0.90, 1.0) * smear;

      // Subtle 8x8 JPEG/MPEG-4 block quantization feel + early CCD white balance
      vec2 blk = floor(fragCoord / 6.0);
      float grain = (hash12(blk + floor(uTime * 12.0)) - 0.5) * 0.032;
      col = col * bayerTint + grain;
      col = mix(vec3(luma), col, 1.14);
      return clamp(col, 0.0, 1.0);
    }
    else if (mode == 1) {
      // 1: VHS Hi-Fi (Y/C chroma bleeding + scanline jitter + tracking band)
      float scan = sin(fragCoord.y * 1.6) * 0.035;
      float trackY = fract(uTime * 0.14);
      float trackBand = smoothstep(0.035, 0.0, abs(uvBlock.y - trackY));
      float snow = hash12(vec2(fragCoord.x * 0.4, fragCoord.y + uTime * 60.0)) * trackBand * 0.35;
      float luma = dot(col, vec3(0.299, 0.587, 0.114));
      col = mix(vec3(luma), col, 0.85) + scan + snow;
      col.r += 0.035;
      col.b += 0.025;
      return clamp(col, 0.0, 1.0);
    }
    else if (mode == 2) {
      // 2: CRT Trinitron (RGB aperture grille triads + scanlines)
      float triad = mod(floor(fragCoord.x), 3.0);
      vec3 mask = triad < 0.5 ? vec3(1.16, 0.88, 0.88) : (triad < 1.5 ? vec3(0.88, 1.16, 0.88) : vec3(0.88, 0.88, 1.16));
      float scanline = 0.90 + 0.10 * sin(fragCoord.y * 3.14159);
      return clamp(col * mask * scanline * 1.05, 0.0, 1.0);
    }
    else if (mode == 3) {
      // 3: GameBoy LCD (4-tone DMG-01 olive-silver matrix with pixel grid)
      float luma = dot(col, vec3(0.299, 0.587, 0.114));
      float q = floor(clamp(luma, 0.0, 0.999) * 4.0) / 3.0;
      vec3 darkOlive = vec3(0.14, 0.21, 0.12);
      vec3 lightOlive = vec3(0.68, 0.75, 0.56);
      vec3 lcd = mix(darkOlive, lightOlive, q);
      vec2 cell = fract(fragCoord / 3.0);
      float grid = smoothstep(0.0, 0.14, cell.x) * smoothstep(0.0, 0.14, cell.y);
      return mix(lcd * 0.82, lcd, grid);
    }
    else if (mode == 4) {
      // 4: Mac 1-Bit (1984 Macintosh Bayer 4x4 ordered dithering, high readability)
      float luma = dot(col, vec3(0.299, 0.587, 0.114));
      float d = step(bayer4(floor(fragCoord * 0.5)), luma * 1.12);
      return mix(vec3(0.22, 0.24, 0.27), vec3(0.93, 0.95, 0.97), d);
    }
    else if (mode == 5) {
      // 5: Digicam HUD (Optical viewfinder brackets, crosshair, EV scale inside glass)
      vec2 ap = abs(uvBlock);
      // Corner AF brackets
      float bracketX = smoothstep(0.005, 0.0, abs(ap.x - 0.18)) * step(0.11, ap.y) * step(ap.y, 0.18);
      float bracketY = smoothstep(0.005, 0.0, abs(ap.y - 0.18)) * step(0.11, ap.x) * step(ap.x, 0.18);
      // Center crosshair
      float crossH = smoothstep(0.003, 0.0, abs(uvBlock.y)) * step(ap.x, 0.035) * step(0.01, ap.x);
      float crossV = smoothstep(0.003, 0.0, abs(uvBlock.x)) * step(ap.y, 0.035) * step(0.01, ap.y);
      float hud = clamp(bracketX + bracketY + crossH + crossV, 0.0, 1.0);
      vec3 hudCol = vec3(0.08, 0.10, 0.12);
      return mix(col, hudCol, hud * 0.65);
    }
    // 6: Pure PMMA Acrylic
    return col;
  }

  void main() {
    // Strictly centered coordinates from gl_FragCoord.xy
    float minDim = min(uResolution.x, uResolution.y);
    vec2 p = (gl_FragCoord.xy - 0.5 * uResolution) / minDim;

    // Subtle 3D tilt parallax shift for the block
    vec2 blockPos = p - uTilt * 0.016;
    float cornerR = 0.042;
    float d = sdRoundBox(blockPos, uBlockHalf, cornerR);

    // Base background outside the acrylic block
    vec3 bgCol = sampleBackground(p, uBg);

    // Soft cast drop shadow beneath the floating acrylic slab
    float shadowD = sdRoundBox(blockPos - vec2(0.012, -0.022) - uTilt * 0.012, uBlockHalf, cornerR);
    float shadow = smoothstep(0.075, -0.01, shadowD) * 0.24;
    bgCol *= (1.0 - shadow);

    if (d > 0.002) {
      // Subtle ambient outer rim caustic
      float outerGlow = exp(-d * 65.0) * 0.14;
      fragColor = vec4(bgCol + vec3(outerGlow), 1.0);
      return;
    }

    // Inside the PMMA Acrylic Block:
    // 1. Analytical 45° chamfered bevel normal
    float bevelWidth = 0.044 * uBevelScale;
    float edgeZone = smoothstep(-bevelWidth, 0.0, d);
    vec2 eps = vec2(0.0015, 0.0);
    vec2 bevelGrad = normalize(vec2(
      sdRoundBox(blockPos + eps.xy, uBlockHalf, cornerR) - sdRoundBox(blockPos - eps.xy, uBlockHalf, cornerR),
      sdRoundBox(blockPos + eps.yx, uBlockHalf, cornerR) - sdRoundBox(blockPos - eps.yx, uBlockHalf, cornerR)
    ) + 1e-5);

    // 2. Non-looping ring-buffer ripples (6 slots)
    vec2 waveOffset = vec2(0.0);
    float waveHighlight = 0.0;
    for (int i = 0; i < 6; i++) {
      vec4 rip = uRipples[i];
      float age = uTime - rip.z;
      if (rip.w > 0.001 && age >= 0.0 && age < 1.8) {
        vec2 diff = blockPos - rip.xy;
        float dist = length(diff);
        float radius = age * 0.42;
        float ring = exp(-pow((dist - radius) * 26.0, 2.0));
        float decay = exp(-age * 2.9) * rip.w;
        float w = sin((dist - radius) * 75.0) * ring * decay;
        vec2 dir = normalize(diff + 1e-5);
        waveOffset += dir * w * 0.018;
        waveHighlight += w * 0.22;
      }
    }

    // 3. Viscous optical dent lens under stationary held finger
    vec2 fingerDiff = blockPos - uFingerPos;
    float fingerDist = length(fingerDiff);
    float lensProfile = exp(-fingerDist * fingerDist * 26.0) * uFingerPress;
    vec2 lensOffset = fingerDiff * lensProfile * 0.22;
    float lensRim = exp(-pow((fingerDist - 0.11) * 24.0, 2.0)) * uFingerPress * 0.18;

    // 4. Snell's law thickness refraction + RGB chromatic dispersion on chamfered edges
    vec2 thicknessParallax = uTilt * 0.028;
    vec2 refractBase = bevelGrad * pow(edgeZone, 1.65) * 0.052 * uBevelScale
                     + thicknessParallax
                     + waveOffset
                     + lensOffset;

    float disp = (0.009 * pow(edgeZone, 1.4) * uBevelScale) + (lensProfile * 0.006);
    float colR = sampleBackground(p - refractBase * (1.0 + disp * 14.0), uBg).r;
    float colG = sampleBackground(p - refractBase, uBg).g;
    float colB = sampleBackground(p - refractBase * (1.0 - disp * 14.0), uBg).b;
    vec3 glassCol = vec3(colR, colG, colB);

    // 5. Internal milky frosted diffusion so #090A0C black text is always crisp & legible
    float lumaBg = dot(glassCol, vec3(0.299, 0.587, 0.114));
    float adaptiveFrost = mix(0.56, 0.30, smoothstep(0.2, 0.85, lumaBg));
    vec3 milkyTint = vec3(0.93, 0.95, 0.98);
    glassCol = mix(glassCol, milkyTint, adaptiveFrost);

    // Subtle PMMA cast acrylic cyan/emerald tint in thick chamfered corners (like 80636.jpg & 80637.jpg)
    float cornerFactor = smoothstep(0.08, 0.32, length(max(abs(blockPos) - (uBlockHalf - 0.09), 0.0)));
    vec3 pmmaTint = vec3(0.06, 0.22, 0.20) * pow(edgeZone, 1.8) * (0.45 + 0.55 * cornerFactor);
    glassCol += pmmaTint;

    // 6. Apply selected retro sensor/display effect
    vec2 localNorm = blockPos / (uBlockHalf * 2.0);
    glassCol = applyRetroEffect(glassCol, localNorm, gl_FragCoord.xy, uEffect);

    // 7. Studio directional specular highlights + chamfer crisp inner/outer edge lines
    vec2 lightDir = normalize(vec2(-0.58, 0.81) + uTilt * 0.45);
    float specBevel = pow(max(0.0, dot(bevelGrad, lightDir)), 5.0) * edgeZone;
    float backRim = pow(max(0.0, dot(bevelGrad, -lightDir)), 4.0) * edgeZone * 0.45;

    // Diagonal studio softbox sheen across the front face
    float sheen = smoothstep(0.32, -0.28, dot(blockPos, vec2(0.65, -0.75)) + uTilt.x * 0.12) * 0.09;

    // Crisp razor chamfer edge line
    float innerChamferLine = smoothstep(0.0028, 0.0, abs(d + bevelWidth * 0.85)) * 0.32;
    float outerRimLine = smoothstep(0.0022, 0.0, abs(d + 0.0012)) * 0.65;

    // Subtle micro-dust / tactile surface texture
    float microGrain = (hash12(floor(gl_FragCoord.xy)) - 0.5) * 0.012;

    glassCol += vec3(specBevel * 0.62 + backRim * 0.35 + sheen + innerChamferLine + outerRimLine + waveHighlight + lensRim + microGrain);

    // Anti-aliased edge blend with background
    float alpha = smoothstep(0.0015, -0.0015, d);
    vec3 finalCol = mix(bgCol, clamp(glassCol, 0.0, 1.0), alpha);

    fragColor = vec4(finalCol, 1.0);
  }`;

  function compileShader(type, source) {
    const shader = gl.createShader(type);
    gl.shaderSource(shader, source);
    gl.compileShader(shader);
    if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) {
      console.error(gl.getShaderInfoLog(shader));
      gl.deleteShader(shader);
      return null;
    }
    return shader;
  }

  const vs = compileShader(gl.VERTEX_SHADER, vsSource);
  const fs = compileShader(gl.FRAGMENT_SHADER, fsSource);
  const program = gl.createProgram();
  gl.attachShader(program, vs);
  gl.attachShader(program, fs);
  gl.linkProgram(program);
  gl.useProgram(program);

  const vao = gl.createVertexArray();
  gl.bindVertexArray(vao);

  // Uniform locations
  const uResolutionLoc = gl.getUniformLocation(program, 'uResolution');
  const uTimeLoc = gl.getUniformLocation(program, 'uTime');
  const uBlockHalfLoc = gl.getUniformLocation(program, 'uBlockHalf');
  const uTiltLoc = gl.getUniformLocation(program, 'uTilt');
  const uEffectLoc = gl.getUniformLocation(program, 'uEffect');
  const uBgLoc = gl.getUniformLocation(program, 'uBg');
  const uBevelScaleLoc = gl.getUniformLocation(program, 'uBevelScale');
  const uRipplesLoc = gl.getUniformLocation(program, 'uRipples');
  const uFingerPosLoc = gl.getUniformLocation(program, 'uFingerPos');
  const uFingerPressLoc = gl.getUniformLocation(program, 'uFingerPress');

  // State
  let currentEffect = 0; // Y2K CCD default
  let currentBg = 0;     // Sky Aura default
  let bevelScale = 1.0;

  let blockHalfX = 0.28;
  let blockHalfY = 0.36;

  // Spring physics for tactile tilt
  let targetTiltX = 0, targetTiltY = 0;
  let tiltX = 0, tiltY = 0;
  let gyroEnabled = false;
  let gyroBaseBeta = null, gyroBaseGamma = null;

  // 6-slot ring buffer for single non-looping ripples
  const MAX_RIPPLES = 6;
  const rippleData = new Float32Array(MAX_RIPPLES * 4);
  let rippleCursor = 0;
  let lastRippleTime = 0;
  let lastRippleX = 0, lastRippleY = 0;

  // Stationary finger lens state
  let isPointerDown = false;
  let fingerX = 0, fingerY = 0;
  let fingerPress = 0;
  let targetFingerPres
  let targetFingerPress = 0;

  // Sync canvas size and exact center-anchored HTML typography face
  function syncLayout() {
    const vw = window.visualViewport ? window.visualViewport.width : window.innerWidth;
    const vh = window.visualViewport ? window.visualViewport.height : window.innerHeight;
    const dpr = Math.min(window.devicePixelRatio || 1, 2.0);

    canvas.width = Math.round(vw * dpr);
    canvas.height = Math.round(vh * dpr);
    canvas.style.width = vw + 'px';
    canvas.style.height = vh + 'px';
    gl.viewport(0, 0, canvas.width, canvas.height);

    const minDim = Math.min(vw, vh);
    // Calculate ideal mobile portrait / tablet block dimensions
    const targetW = Math.min(vw * 0.74, 310);
    const targetH = Math.min(vh * 0.54, 420);

    blockHalfX = (targetW * 0.5) / minDim;
    blockHalfY = (targetH * 0.5) / minDim;

    faceEl.style.width = Math.round(targetW) + 'px';
    faceEl.style.height = Math.round(targetH) + 'px';
  }

  syncLayout();
  window.addEventListener('resize', syncLayout, { passive: true });
  if (window.visualViewport) {
    window.visualViewport.addEventListener('resize', syncLayout, { passive: true });
  }
  if ('ResizeObserver' in window) {
    new ResizeObserver(syncLayout).observe(document.body);
  }

  function screenToLocal(clientX, clientY) {
    const vw = window.innerWidth;
    const vh = window.innerHeight;
    const minDim = Math.min(vw, vh);
    const px = (clientX - vw * 0.5) / minDim;
    const py = -(clientY - vh * 0.5) / minDim;    return { x: px, y: py };
  }

  function spawnRipple(lx, ly, nowSec, strength = 1.0) {
    const base = rippleCursor * 4;
    rippleData[base] = lx;
    rippleData[base + 1] = ly;
    rippleData[base + 2] = nowSec;
    rippleData[base + 3] = strength;
    rippleCursor = (rippleCursor + 1) % MAX_RIPPLES;
    lastRippleTime = nowSec;
    lastRippleX = lx;
    lastRippleY = ly;
  }

  // Pointer / Touch interaction (non-looping ripples + viscous lens on hold)
  canvas.addEventListener('pointerdown', (e) => {
    isPointerDown = true;
    const pos = screenToLocal(e.clientX, e.clientY);
    fingerX = pos.x;
    fingerY = pos.y;
    targetFingerPress = 1.0;

    const nowSec = performance.now() * 0.001;
    spawnRipple(pos.x, pos.y, nowSec, 1.0);

    targetTiltX = Math.max(-1, Math.min(1, pos.x / Math.max(blockHalfX, 0.1)));
    targetTiltY = Math.max(-1, Math.min(1, pos.y / Math.max(blockHalfY, 0.1)));
  }, { passive: true });

  window.addEventListener('pointermove', (e) => {
    const pos = screenToLocal(e.clientX, e.clientY);
    if (isPointerDown) {
      fingerX = pos.x;
      fingerY = pos.y;
      targetTiltX = Math.max(-1, Math.min(1, pos.x / Math.max(blockHalfX, 0.1)));
      targetTiltY = Math.max(-1, Math.min(1, pos.y / Math.max(blockHalfY, 0.1)));

      const nowSec = performance.now() * 0.001;
      const moveDist = Math.hypot(pos.x - lastRippleX, pos.y - lastRippleY);
      // Spawn a single new ripple ONLY when dragging past threshold + cooldown (never loops while holding)
      if (moveDist > 0.055 && (nowSec - lastRippleTime) > 0.095) {
        spawnRipple(pos.x, pos.y, nowSec, 0.75);
      }
    } else if (!gyroEnabled && e.pointerType === 'mouse') {
      targetTiltX = Math.max(-1, Math.min(1, pos.x * 1.4));
      targetTiltY = Math.max(-1, Math.min(1, pos.y * 1.4));
    }
  }, { passive: true });

  const releasePointer = () => {
    isPointerDown = false;
    targetFingerPress = 0.0;
    if (!gyroEnabled) {
      targetTiltX = 0;
      targetTiltY = 0;
    }
  };
  window.addEventListener('pointerup', releasePointer, { passive: true });
  window.addEventListener('pointercancel', releasePointer, { passive: true });

  // Gyroscope support for Android phones
  const gyroBtn = document.getElementById('gyroBtn');
  function handleOrientation(e) {
    if (!gyroEnabled || e.beta === null || e.gamma === null) return;
    if (gyroBaseBeta === null) {
      gyroBaseBeta = e.beta;
      gyroBaseGamma = e.gamma;
    }
    const dGamma = Math.max(-30, Math.min(30, e.gamma - gyroBaseGamma)) / 30;
    const dBeta = Math.max(-30, Math.min(30, e.beta - gyroBaseBeta)) / 30;
    targetTiltX = dGamma;
    targetTiltY = -dBeta;
  }

  gyroBtn.addEventListener('click', async () => {
    if (gyroEnabled) {
      gyroEnabled = false;
      gyroBaseBeta = null;
      gyroBaseGamma = null;
      targetTiltX = 0;
      targetTiltY = 0;
      gyroBtn.classList.remove('active');
      return;
    }
    if (typeof DeviceOrientationEvent !== 'undefined' && typeof DeviceOrientationEvent.requestPermission === 'function') {
      try {
        const res = await DeviceOrientationEvent.requestPermission();
        if (res !== 'granted') return;
      } catch (_) {
        return;
      }
    }
    gyroEnabled = true;
    gyroBaseBeta = null;
    gyroBaseGamma = null;
    gyroBtn.classList.add('active');
    window.addEventListener('deviceorientation', handleOrientation, { passive: true });
  });

  // UI Chip selectors
  document.getElementById('effectRow').addEventListener('click', (e) => {
    const btn = e.target.closest('.chip');
    if (!btn) return;
    document.querySelectorAll('#effectRow .chip').forEach(c => c.classList.remove('active'));
    btn.classList.add('active');
    currentEffect = parseInt(btn.dataset.effect, 10);
  });

  document.getElementById('bgRow').addEventListener('click', (e) => {
    const btn = e.target.closest('.chip');
    if (!btn) return;
    document.querySelectorAll('#bgRow .chip').forEach(c => c.classList.remove('active'));
    btn.classList.add('active');
    currentBg = parseInt(btn.dataset.bg, 10);
  });

  const bevelSlider = document.getElementById('bevelSlider');
  const bevelVal = document.getElementById('bevelVal');
  bevelSlider.addEventListener('input', () => {
    bevelScale = parseFloat(bevelSlider.value);
    bevelVal.textContent = bevelScale.toFixed(2) + 'x';
  });

  // AGSL Modal controls
  const agslModal = document.getElementById('agslModal');
  const agslBtn = document.getElementById('agslBtn');
  const closeAgslBtn = document.getElementById('closeAgslBtn');
  const copyAgslBtn = document.getElementById('copyAgslBtn');
  const agslCode = document.getElementById('agslCode');

  agslBtn.addEventListener('click', () => agslModal.classList.add('open'));
  closeAgslBtn.addEventListener('click', () => agslModal.classList.remove('open'));
  agslModal.addEventListener('click', (e) => {
    if (e.target === agslModal) agslModal.classList.remove('open');
  });

  copyAgslBtn.addEventListener('click', async () => {
    try {
      await navigator.clipboard.writeText(agslCode.textContent);
      copyAgslBtn.textContent = 'Скопировано!';
      setTimeout(() => { copyAgslBtn.textContent = 'Скопировать AGSL код'; }, 1600);
    } catch (_) {
      copyAgslBtn.textContent = 'Выделите текст выше';
    }
  });

  // Render loop
  function render(nowMs) {
    const nowSec = nowMs * 0.001;

    // Smooth spring damping
    tiltX += (targetTiltX - tiltX) * 0.14;
    tiltY += (targetTiltY - tiltY) * 0.14;
    fingerPress += (targetFingerPress - fingerPress) * 0.16;

    // Sync crisp black typography layer 3D perspective with shader slab
    const rotY = (tiltX * 5.5).toFixed(2);
    const rotX = (-tiltY * 5.5).toFixed(2);
    const shiftX = (tiltX * 5.0).toFixed(2);
    const shiftY = (-tiltY * 5.0).toFixed(2);
    faceEl.style.transform = `translate(calc(-50% + ${shiftX}px), calc(-50% + ${shiftY}px)) perspective(900px) rotateX(${rotX}deg) rotateY(${rotY}deg)`;

    gl.uniform2f(uResolutionLoc, canvas.width, canvas.height);
    gl.uniform1f(uTimeLoc, nowSec);
    gl.uniform2f(uBlockHalfLoc, blockHalfX, blockHalfY);
    gl.uniform2f(uTiltLoc, tiltX, tiltY);
    gl.uniform1i(uEffectLoc, currentEffect);
    gl.uniform1i(uBgLoc, currentBg);
    gl.uniform1f(uBevelScaleLoc, bevelScale);
    gl.uniform4fv(uRipplesLoc, rippleData);
    gl.uniform2f(uFingerPosLoc, fingerX, fingerY);
    gl.uniform1f(uFingerPressLoc, fingerPress);

    gl.drawArrays(gl.TRIANGLES, 0, 3);
    requestAnimationFrame(render);
  }

  requestAnimationFrame(render);
})();
